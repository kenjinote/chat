   464	- (void)showLocalVideoWindowWithTitleSuffix:(NSString *)suffix {
   465	    std::string uname = g_localUserName;
   466	    NSString *rawName = [NSString stringWithUTF8String:uname.c_str()];
   467	    NSRange hashRange = [rawName rangeOfString:@"#"];
   468	    NSString *dispName = (hashRange.location != NSNotFound) ? [rawName substringToIndex:hashRange.location] : rawName;
   469	    
   470	    if (suffix) {
   471	        dispName = [dispName stringByAppendingString:suffix];
   472	    }
   473	    
   474	    if (g_videoWindows.find(g_userId) == g_videoWindows.end()) {
   475	        VideoWindowController *videoWin = [[VideoWindowController alloc] initWithUserId:[NSString stringWithUTF8String:g_userId.c_str()] userName:dispName];

   477	        videoWin.window.title = dispName;
   478	        g_videoWindows[g_userId] = videoWin;
   479	    } else {
   480	        [g_videoWindows[g_userId] showWindow:self];
   481	        [g_videoWindows[g_userId].window makeKeyAndOrderFront:self];
   482	        g_videoWindows[g_userId].window.title = dispName;
   483	    }
   484	    if (g_captureSourceType == 1) {

   486	        self.videoButton.title = [NSString stringWithFormat:@"%@: OFF", NSLocalizedString(@"IDS_VIDEO_STREAM", nil)];

   489	            [g_videoWindows[g_userId] close];
   490	            g_videoWindows.erase(g_userId);
   491	        }
   492	    } else {
   493	        g_captureSourceType = 1;
   494	        self.videoButton.title = [NSString stringWithFormat:@"%@: ON", NSLocalizedString(@"IDS_VIDEO_STREAM", nil)];
   495	        [self showLocalVideoWindowWithTitleSuffix:nil];
